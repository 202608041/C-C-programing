문제 1. 다음 코드의 실행 결과로 올바른 것은?


#include <stdio.h>

struct Student {
    char name[20];
    int age;
};

void changeAge(struct Student *s) {
    s->age = 25;
}

int main() {
    struct Student s1 = {"Kim", 20};

    changeAge(&s1);

    printf("%d", s1.age);

    return 0;
}



① 20
② 25
③ 0
④ 오류 발생
